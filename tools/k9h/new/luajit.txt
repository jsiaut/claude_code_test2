local function run_k9(v, reps)
  -- harnais commun de K9 (docs/K9.md) : grilles allouées une fois par processus, réinitialisées hors chrono
  local ga, gb
  if v == "natural" then ga, gb = {}, {}; for i = 1, M9 do ga[i], gb[i] = {}, {} end
  else ga, gb = ffi.new("double[?]", M9 * M9), ffi.new("double[?]", M9 * M9) end
  for r = -WARMUP, reps - 1 do
    local s, dt = 0.0, 0
    if v == "natural" then
      local a, b = ga, gb
      for i = 1, M9 do for j = 1, M9 do local x = k9_init(i - 1, j - 1); a[i][j] = x; b[i][j] = x end end
      local t0 = clock()
      k9_natural(a, b); k9_natural(b, a); k9_natural(a, b); k9_natural(b, a)
      dt = clock() - t0
      for i = 1, M9 do local row = a[i]; for j = 1, M9 do s = s + row[j] end end
    else
      local a, b = ga, gb
      for k = 0, M9 * M9 - 1 do local x = k9_init(math.floor(k / M9), k % M9); a[k] = x; b[k] = x end
      local t0 = clock()
      k9_flat(a, b); k9_flat(b, a); k9_flat(a, b); k9_flat(b, a)
      dt = clock() - t0
      for k = 0, M9 * M9 - 1 do s = s + a[k] end
    end
    if r >= 0 then result("k9", v, "2048^2x4", r, dt, string.format("%.15e", s)) end
  end
end
